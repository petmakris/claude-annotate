# deny.txt: what scan.py refuses in a clip's sources, frames and captions.
#
#   re: <regex>     a generic pattern, matched without case
#   allow: <regex>  text cut out before matching (invented addresses and ids)
#   h: <hash>       a private word (a person, a company, a host, a ticket prefix),
#                   kept only as a salted hash so this public file never spells it.
#                   Add one with: uv run tools/promo/scan.py hash <word>
#
# Words only one machine should know go in a file outside the repo named by
# $PROMO_DENY_EXTRA, one per line.

allow: [a-z0-9._%+-]+@example\.(com|org|net)
allow: \b127\.0\.0\.1\b|\b0\.0\.0\.0\b

# Addresses, keys and tokens
re: [a-z0-9._%+-]+@[a-z0-9.-]+\.[a-z]{2,}
re: \b(?:\d{1,3}\.){3}\d{1,3}\b
re: \bghp_[A-Za-z0-9]{20,}
re: \bgh[ousr]_[A-Za-z0-9]{20,}
re: \bsk-[A-Za-z0-9_-]{16,}
re: \bxox[abposr]-[A-Za-z0-9-]{8,}
re: \bAKIA[0-9A-Z]{16}\b
re: \beyJ[A-Za-z0-9_-]{8,}\.[A-Za-z0-9_-]{8,}
re: -----BEGIN [A-Z ]*PRIVATE KEY-----

# Session links, attribution, real home folders
re: claude\.ai/code/session
re: co-authored-by:\s*claude
re: /Users/[A-Za-z]
re: ~/projects/
re: \b[a-z0-9-]+\.mac\b
re: \.ts\.net\b

# Ports of services a person runs every day (a clip never shows them)
re: (?<![\d.]):(3080|3081|8766|8770|5433)\b
